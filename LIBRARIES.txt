Y = read.zoo(
  data.frame(df),
  FUN = as.Date,
  format = "%d%m%Y"
)
head(Y)
plot(Y)
library(FinTS)

adf.test(Y$Nifty)

pp.test(Y$Gold)

returns <-diff(log(Y))

ArchTest(returns, lags = 10)
Box.test(returns$Nifty, lag = 10, type = "Ljung-Box")
Box.test(returns$Gold, lag = 10, type = "Ljung-Box")
Box.test(returns$Crude, lag = 10, type = "Ljung-Box")
jarque.bera.test(returns$Nifty)
jarque.bera.test(returns$Gold)
jarque.bera.test(returns$Crude)

library(rugarch)

spec = ugarchspec(
  variance.model = list(model="sGARCH", garchOrder=c(1,1)),
  mean.model = list(armaOrder=c(0,0)),
  distribution.model = "std"
)
fit1 <-ugarchfit(spec = spec, data = returns$Nifty)
fit2 <-ugarchfit(spec = spec, data = returns$Gold)
fit3 <-ugarchfit(spec = spec, data = returns$Crude)

res1 <-residuals(fit1)
res2 <-residuals(fit2)
res3 <-residuals(fit3)

Box.test(res1, lag = 10, type = "Ljung-Box")
Box.test(res2, lag = 10, type = "Ljung-Box")
Box.test(res3, lag = 10, type = "Ljung-Box")

ArchTest(res1, lags = 10)
ArchTest(res2, lags = 10)
ArchTest(res3, lags = 10)

forecast1 <-ugarchforecast(fit1, n.ahead = 10)
forecast2 <-ugarchforecast(fit2, n.ahead = 10)
forecast3 <-ugarchforecast(fit3, n.ahead = 10)

fit_egarch
