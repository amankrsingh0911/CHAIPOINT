import pandas as pd 
from sklearn.model_selection import train_test_split 
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score
from sklearn.naive_bayes import MultinomialNB
from sklearn.feature_extraction.text import TfidfVectorizer

df = pd.DataFrame(data)
x = df["text"]
y = df["sentiment"]
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=42)
vectorize = TfidfVectorizer(stop_words= "english", lowercase=True)
x_train_vec = vectorize.fit_transform(x_train)
x_test_vec = vectorize.transform(x_test)
lr_model = LogisticRegression()
lr_model.fit(x_train_vec, y_train)
nb_model = MultinomialNB()
nb_model.fit(x_train_vec, y_train)
lr_preds= lr_model.predict(x_test_vec)
nb_preds=nb_model.predict(x_test_vec)
accuracy_lr = accuracy_score(y_test, lr_preds)
accuracy_nb = accuracy_score(y_test, nb_preds)
def predict_sentiment(new_texts):
    new_texts_vec = vectorize.transform(new_texts)
    lr_preds = lr_model.predict(new_texts_vec)
    nb_preds = nb_model.predict(new_texts_vec)
    return lr_preds, nb_preds

custom_samples = [
    "The item arrived broken and late.",
    "It is an amazing product.",
    "Not worth the money."
]

lr_preds, nb_preds = predict_sentiment(custom_samples)
print("Logistic Regression:", lr_preds)
print("Naive Bayes:", nb_preds)

import pandas as pd
from sklearn import datasets
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
from sklearn.metrics import accuracy_score, classification_report

iris=datasets.load_iris()
x=iris.data
y=iris.target
x_train,x_test,y_train,y_test=train_test_split(x,y,test_size=0.2, random_state=42, stratify=y)
scalar=StandardScaler()
x_train_scaled=scalar.fit_transform(x_train)
x_test_scaled=scalar.fit_transform(x_test)
evaluation_model=SVC(kernel="rbf",C=1.0, random_state=42)
evaluation_model.fit(x_train_scaled, y_train)
y_pred=evaluation_model.predict(x_test_scaled)
accuracy=accuracy_score(y_test, y_pred)
print("=== SVM model evaluation report ===")
